      272     must_have = question_schema.must_have_terms

      281     matched_in_behavior = sum(1 for term in must_have if term in behavior)

      284     coverage = (matched_in_impact * 2 + matched_in_behavior) / len(must_have)

      286     if coverage >= 0.8:  # 涵蓋 80% 以上
